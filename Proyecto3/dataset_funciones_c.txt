int ES_PRIMO(int v_numero)
{
    int v_i;
    if (v_numero <= 1)
    {
        return 0;
    }
    for (v_i = 2; v_i < v_numero; v_i++)
    {
        if (v_numero % v_i == 0)
        {
            return 0;
        }
    }
    return 1;
}

int ES_NUMERO_PERFECTO(int v_numero)
{
    int v_suma = 0;
    int v_i;
    for (v_i = 1; v_i < v_numero; v_i++)
    {
        if (v_numero % v_i == 0)
        {
            v_suma += v_i;
        }
    }
    return v_suma == v_numero;
}

{
    if (v_tamano <= 0)
    {
        return 0.0f;
    }
    int v_suma = SUMAR_ARREGLO(v_arreglo, v_tamano);

// [CRIS_CODE]
void ORDENAR_BURBUJA(int v_arreglo[], int v_tamano)
{
    int v_i;
    int v_j;
    for (v_i = 0; v_i < v_tamano - 1; v_i++)
    {
        for (v_j = 0; v_j < v_tamano - v_i - 1; v_j++)
        {
            if (v_arreglo[v_j] > v_arreglo[v_j + 1])
            {
                INTERCAMBIAR_VALORES(&v_arreglo[v_j], &v_arreglo[v_j + 1]);
            }
        }
    }
}



// [CRIS_CODE]
void INVERTIR_ARREGLO(int v_arreglo[], int v_tamano)
{
    int v_inicio = 0;
    int v_fin = v_tamano - 1;
    while (v_inicio < v_fin)
    {
        INTERCAMBIAR_VALORES(&v_arreglo[v_inicio], &v_arreglo[v_fin]);
        v_inicio++;
        v_fin--;
    }
}

// [CRIS_CODE]
void MULTIPLICAR_MATRICES_2X2(int v_m1[2][2], int v_m2[2][2], int v_resultado[2][2])
{
    int v_i;
    int v_j;
    int v_k;
    for (v_i = 0; v_i < 2; v_i++)
    {
        for (v_j = 0; v_j < 2; v_j++)
        {
            v_resultado[v_i][v_j] = 0;
            for (v_k = 0; v_k < 2; v_k++)
            {
                v_resultado[v_i][v_j] += v_m1[v_i][v_k] * v_m2[v_k][v_j];
            }
        }
    }
}



// [CRIS_CODE]
void SUMAR_MATRICES_2X2(int v_m1[2][2], int v_m2[2][2], int v_resultado[2][2])
{
    int v_i;
    int v_j;
    for (v_i = 0; v_i < 2; v_i++)
    {
        for (v_j = 0; v_j < 2; v_j++)
        {
            v_resultado[v_i][v_j] = v_m1[v_i][v_j] + v_m2[v_i][v_j];
        }
    }
}

int LONGITUD_CADENA(char v_cadena[])
{
    int v_longitud = 0;
    while (v_cadena[v_longitud] != '\0')
    {
        v_longitud++;
    }
    return v_longitud;
}



// [CRIS_CODE]
void COPIAR_CADENA(char v_origen[], char v_destino[])
{
    int v_i = 0;
    while (v_origen[v_i] != '\0')
    {
        v_destino[v_i] = v_origen[v_i];
        v_i++;
    }
    v_destino[v_i] = '\0';
}



// [CRIS_CODE]
void CONCATENAR_CADENA(char v_destino[], char v_origen[])
{
    int v_i = LONGITUD_CADENA(v_destino);
    int v_j = 0;
    while (v_origen[v_j] != '\0')
    {
        v_destino[v_i] = v_origen[v_j];
        v_i++;
        v_j++;
    }
    v_destino[v_i] = '\0';
}



// [CRIS_CODE]
void INVERTIR_CADENA(char v_cadena[])
{
    int v_inicio = 0;
    int v_fin = LONGITUD_CADENA(v_cadena) - 1;
    char v_temp;
    while (v_inicio < v_fin)
    {
        v_temp = v_cadena[v_inicio];
        v_cadena[v_inicio] = v_cadena[v_fin];
        v_cadena[v_fin] = v_temp;
        v_inicio++;
        v_fin--;
    }
}

int ES_PALINDROMO(char v_cadena[])
{

    while (v_inicio < v_fin)

// [CRIS_CODE]
int COMPARAR_CADENAS(char v_cad1[], char v_cad2[])
{
    int v_i = 0;
    while (v_cad1[v_i] != '\0' && v_cad2[v_i] != '\0')
    {
        if (v_cad1[v_i] != v_cad2[v_i])
        {
            return v_cad1[v_i] - v_cad2[v_i];
        }
        v_i++;
    }
    return v_cad1[v_i] - v_cad2[v_i];
}

int CONTAR_CONSONANTES(char v_cadena[])
{
    int v_contador = 0;
    int v_i = 0;
    char v_caracter;
    while (v_cadena[v_i] != '\0')
    {
        v_caracter = tolower(v_cadena[v_i]);
        if (isalpha(v_caracter) && v_caracter != 'a' && v_caracter != 'e' && v_caracter != 'i' && v_caracter != 'o' && v_caracter != 'u')
        {
            v_contador++;
        }
        v_i++;
    }
    return v_contador;
}

void ELIMINAR_ESPACIOS(char v_origen[], char v_destino[])
{
    int v_i = 0;
    int v_j = 0;
    while (v_origen[v_i] != '\0')
    {
        if (v_origen[v_i] != ' ')
        {
            v_destino[v_j] = v_origen[v_i];
            v_j++;

    }
    v_destino[v_j] = '\0';
}

int CONTAR_PALABRAS(char v_cadena[])
{
    int v_contador = 0;
    int v_i = 0;
    int v_en_palabra = 0;
    while (v_cadena[v_i] != '\0')
    {
        if (v_cadena[v_i] == ' ' || v_cadena[v_i] == '\t' || v_cadena[v_i] == '\n')
        {
            v_en_palabra = 0;
        }
        else if (v_en_palabra == 0)
        {
            v_en_palabra = 1;
            v_contador++;
        }
        v_i++;
    }
    return v_contador;
}

{

// [CRIS_CODE]
float DISTANCIA_PUNTOS(Punto2D v_p1, Punto2D v_p2)
{
    float v_dx = v_p2.v_x - v_p1.v_x;
    float v_dy = v_p2.v_y - v_p1.v_y;
    return sqrtf(v_dx * v_dx + v_dy * v_dy);
}

Nodo* CREAR_NODO(int v_dato)
{
    Nodo *v_nuevo = (Nodo*)malloc(sizeof(Nodo));
    if (v_nuevo != NULL)
    {
        v_nuevo->v_dato = v_dato;
        v_nuevo->v_siguiente = NULL;
    }
    return v_nuevo;
}



// [CRIS_CODE]
Nodo* INSERTAR_NODO_FINAL(Nodo *v_cabeza, int v_dato)
{
    Nodo *v_nuevo = CREAR_NODO(v_dato);
    if (v_cabeza == NULL)
    {
        return v_nuevo;
    }
    Nodo *v_temp = v_cabeza;
    while (v_temp->v_siguiente != NULL)
    {
        v_temp = v_temp->v_siguiente;
    }
    v_temp->v_siguiente = v_nuevo;
    return v_cabeza;
}



// [CRIS_CODE]
int CONTAR_NODOS(Nodo *v_cabeza)
{
    int v_contador = 0;
    Nodo *v_temp = v_cabeza;
    while (v_temp != NULL)
    {
        v_contador++;
        v_temp = v_temp->v_siguiente;
    }
    return v_contador;
}



// [CRIS_CODE]
Nodo* BUSCAR_NODO(Nodo *v_cabeza, int v_buscar)
{
    Nodo *v_temp = v_cabeza;
    while (v_temp != NULL)
    {
        if (v_temp->v_dato == v_buscar)
        {
            return v_temp;
        }
        v_temp = v_temp->v_siguiente;
    }
    return NULL;
}

// [CRIS_CODE]
int SUMAR_DIGITOS(int v_numero)
{
    int v_suma = 0;
    int v_temp = v_numero;
    if (v_temp < 0)
    {
        v_temp = -v_temp;
    }
    while (v_temp > 0)
    {
        v_suma += v_temp % 10;
        v_temp /= 10;
    }
    return v_suma;
}
